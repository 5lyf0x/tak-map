# --- TAK Map v1.4.0 i620 ADS-B provider priority / failover hotfix ---
# Keeps the i465/i491 single background collector and existing browser worker.
# Failover is SERVER-SIDE, so cross-origin browser restrictions do not apply.
try:
    import json as _adsbfo_json
    import math as _adsbfo_math
    import time as _adsbfo_time
    from urllib.request import Request as _adsbfo_Request, urlopen as _adsbfo_urlopen

    _ADSBFO_PRIORITY = ('airplanes.live', 'adsb.lol', 'adsb.fi')
    _ADSBFO_STATUS = {
        name: {'failures': 0, 'retry_at': 0.0, 'last_error': '',
               'last_success_at': 0.0, 'last_http_status': 0, 'last_attempt_at': 0.0}
        for name in _ADSBFO_PRIORITY
    }
    _ADSBFO_ACTIVE = ''
    _ADSBFO_USER_AGENT = 'TAKMap/1.4.0 ADSB-multi-source (aircraft tracking; contact project administrator)'

    def _adsbfo_url(name, settings):
        try:
            lat = float(settings['center_lat'])
            lon = float(settings['center_lon'])
            miles = float(settings.get('radius_miles') or 50)
            if not (-90 <= lat <= 90 and -180 <= lon <= 180 and _adsbfo_math.isfinite(miles)):
                raise ValueError('invalid center or radius')
            # Sidebar uses statute miles; provider APIs require nautical miles.
            nm = min(250, max(1, int(_adsbfo_math.ceil(miles / 1.150779448))))
        except (KeyError, ValueError, TypeError, OverflowError):
            raise ValueError('ADS-B center/radius is not configured')
        center = f'{lat:.6f}/{lon:.6f}'
        if name == 'airplanes.live':
            return f'https://api.airplanes.live/v2/point/{center}/{nm}'
        if name == 'adsb.lol':
            return f'https://api.adsb.lol/v2/point/{center}/{nm}'
        if name == 'adsb.fi':
            return f'https://opendata.adsb.fi/api/v3/lat/{lat:.6f}/lon/{lon:.6f}/dist/{nm}'
        raise ValueError('Unsupported ADS-B provider')

    def _adsbfo_json_get(name, url):
        headers = {'Accept': 'application/json', 'User-Agent': _ADSBFO_USER_AGENT}
        pool = globals().get('_I465_ADSB_POOL')
        if pool is not None:
            timeout_type = getattr(globals().get('_i465_urllib3'), 'Timeout', None)
            options = {'connect': 2.5, 'read': 4.0}
            req_kwargs = {'headers': headers, 'preload_content': True,
                          'redirect': False, 'retries': False}
            if timeout_type is not None:
                req_kwargs['timeout'] = timeout_type(**options)
            resp = pool.request('GET', url, **req_kwargs)
            try:
                status = int(resp.status)
                if status != 200:
                    raise RuntimeError(f'HTTP {status}')
                raw = bytes(resp.data or b'')
            finally:
                resp.release_conn()
        else:
            from urllib.error import HTTPError as _adsbfo_HTTPError
            req = _adsbfo_Request(url, headers=headers)
            try:
                with _adsbfo_urlopen(req, timeout=6.5) as resp:
                    raw = resp.read(8 * 1024 * 1024 + 1)
            except _adsbfo_HTTPError as exc:
                raise RuntimeError(f'HTTP {exc.code}') from exc
        if len(raw) > 8 * 1024 * 1024:
            raise RuntimeError('Response exceeds 8 MB')
        try:
            doc = _adsbfo_json.loads(raw.decode('utf-8'))
        except (UnicodeDecodeError, ValueError) as exc:
            raise RuntimeError('Invalid JSON') from exc
        if not isinstance(doc, dict) or not (isinstance(doc.get('ac'), list) or isinstance(doc.get('aircraft'), list)):
            raise RuntimeError('Invalid aircraft response (missing ac/aircraft array)')
        # ac: [] is a valid answer and must never trigger failover.
        return doc

    def _adsbfo_cooldown(error, failures):
        msg = str(error)
        if 'HTTP 401' in msg or 'HTTP 403' in msg:
            return 300.0  # Access restrictions / feeder-only API
        if 'HTTP 429' in msg:
            return 90.0   # Provider rate-limited
        return float(min(90, 10 * (2 ** min(failures - 1, 3))))

    def _adsbfo_collect_once(settings, key):
        global _ADSBFO_ACTIVE
        # Reject invalid location before penalizing individual providers.
        _adsbfo_url(_ADSBFO_PRIORITY[0], settings)
        now = _adsbfo_time.time()
        attempts = []
        for name in _ADSBFO_PRIORITY:
            with _I464_ADSB_LOCK:
                state = _ADSBFO_STATUS[name]
                retry_at = float(state['retry_at'])
            if now < retry_at:
                attempts.append({'source': name, 'ok': False, 'skipped': True,
                                 'retry_seconds': round(retry_at - now, 1)})
                continue
            # Respect the strict 1 request/second public rate limit on ADSB.fi
            # even when the user changes the map center rapidly.
            if name == 'adsb.fi':
                with _I464_ADSB_LOCK:
                    since = _adsbfo_time.time() - float(_ADSBFO_STATUS[name]['last_attempt_at'])
                if since < 1.1:
                    _adsbfo_time.sleep(1.1 - max(0.0, since))
            with _I464_ADSB_LOCK:
                _ADSBFO_STATUS[name]['last_attempt_at'] = _adsbfo_time.time()
            url = _adsbfo_url(name, settings)
            began = _adsbfo_time.monotonic()
            try:
                data = _adsbfo_json_get(name, url)
                rows, raw_count, source_now, stale, sig = _i465_adsb_normalize(data, settings)
                # Normalize once, then keep all existing markers, filters and
                # interpolated track identity. Only provenance changes.
                display = {'airplanes.live': 'Airplanes.live',
                           'adsb.lol': 'ADSB.lol', 'adsb.fi': 'ADSB.fi'}[name]
                for aircraft in rows:
                    aircraft['source'] = display
                elapsed = _adsbfo_time.monotonic() - began
                with _I464_ADSB_LOCK:
                    state = _ADSBFO_STATUS[name]
                    state.update({'failures': 0, 'retry_at': 0.0, 'last_error': '',
                                  'last_success_at': _adsbfo_time.time(), 'last_http_status': 200})
                    _ADSBFO_ACTIVE = name
                    _I464_ADSB_STATE.update({'fetch_seconds': elapsed,
                                             'fetch_count': int(_I464_ADSB_STATE.get('fetch_count') or 0) + 1,
                                             'last_success_source_at': _adsbfo_time.time(),
                                             'active_provider': name})
                attempts.append({'source': name, 'ok': True,
                                 'request_seconds': round(elapsed, 3)})
                note = {'source': name, 'ok': True, 'url': url,
                        'background': True, 'request_seconds': round(elapsed, 3),
                        'raw_count': raw_count, 'attempts': attempts,
                        'raw_array_field': _I464_ADSB_STATE.get('raw_array_field') or '',
                        'top_level_keys': list(_I464_ADSB_STATE.get('top_level_keys') or [])}
                _i464_adsb_store(key, settings, rows, raw_count,
                                 source_now, stale, sig, note)
                return
            except Exception as exc:
                msg = str(exc)[:200]
                with _I464_ADSB_LOCK:
                    state = _ADSBFO_STATUS[name]
                    failures = int(state.get('failures') or 0) + 1
                    state.update({'failures': failures,
                                  'retry_at': _adsbfo_time.time() + _adsbfo_cooldown(msg, failures),
                                  'last_error': msg,
                                  'last_http_status': int(msg.split()[1]) if msg.startswith('HTTP ') and msg.split()[1].isdigit() else 0})
                attempts.append({'source': name, 'ok': False, 'error': msg})
        # Preserve existing last-good snapshot on total provider failure.
        summary = ', '.join(f"{x['source']}: {x.get('error') or 'cooldown'}" for x in attempts)
        raise RuntimeError('All ADS-B providers unavailable (' + summary + ')')

    # The existing collector looks up this function dynamically each cycle.
    _i464_adsb_collect_once = _adsbfo_collect_once

    _adsbfo_base_adsb = tak_map_adsb
    def tak_map_adsb(query_settings=None):
        out = _adsbfo_base_adsb(query_settings)
        if not isinstance(out, dict):
            return out
        with _I464_ADSB_LOCK:
            snapshot = _I464_ADSB_SNAPSHOTS.get(out.get('snapshot_key')) or {}
            note = dict(snapshot.get('source_note') or {})
            current = str(note.get('source') or _ADSBFO_ACTIVE or '')
            health = [dict({'source': name}, **_ADSBFO_STATUS[name]) for name in _ADSBFO_PRIORITY]
            active_error = str(_I464_ADSB_STATE.get('last_error') or '')
        # The existing browser worker treats sources[].ok=false as a map-wide
        # outage. Only the selected provider belongs in that array; detailed
        # failover diagnostics remain separate in provider_status.
        out['sources'] = [{'source': current or 'unavailable',
                           'ok': bool(current and not active_error),
                           'background': True,
                           'snapshot': bool(snapshot),
                           'error': active_error}]
        out['active_source'] = current or None
        out['source_priority'] = list(_ADSBFO_PRIORITY)
        out['provider_status'] = health
        out['provider_attempts'] = list(note.get('attempts') or [])
        out['source_mode'] = 'priority_failover_background_snapshot'
        return out
    globals()['tak_map_adsb'] = tak_map_adsb

    _adsbfo_base_config = tak_map_config
    def tak_map_config():
        cfg = _adsbfo_base_config()
        if isinstance(cfg, dict):
            adsb = dict(cfg.get('adsb') or {})
            adsb.update({'source_mode': 'priority_failover_background_snapshot',
                         'source_priority': list(_ADSBFO_PRIORITY),
                         'source_failover': True})
            cfg['adsb'] = adsb
        return cfg
    globals()['tak_map_config'] = tak_map_config
except Exception as _adsbfo_error:
    raise RuntimeError('TAK Map ADS-B failover patch failed to load') from _adsbfo_error
# --- end TAK Map v1.4.0 i620 ADS-B provider priority / failover hotfix ---
